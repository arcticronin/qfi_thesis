\chapter{Mathematical Background}
\label{ch:MathBackground}

This chapter introduces only the mathematical tools used in the remainder of the thesis: mixed quantum states, Fisher information, the symmetric logarithmic derivative, fidelity-based QFI bounds, and the quantum natural gradient. Supplementary material on generalized measurements, geometric extensions, and multiparameter bounds is collected in Appendix~\ref{app:mathematical_background}.

\section{Quantum States, Mixedness, and Open Systems}

\subsection{Density Operators, Purity, and Ontic versus Epistemic Uncertainty}

\begin{definition}[Density operator and purity]
A quantum state is represented by a density operator $\rho$ on a Hilbert space $\mathcal{H}$~\cite{Sakurai_Napolitano_2020}:
\begin{equation}
    \rho^\dagger = \rho, \qquad \rho \geq 0, \qquad \operatorname{Tr}(\rho)=1.
\end{equation}
Its spectral decomposition is
\begin{equation}
    \rho = \sum_k p_k |\psi_k\rangle\langle\psi_k|, \qquad p_k\geq0,\qquad \sum_k p_k=1,
\end{equation}
and its purity is
\begin{equation}
    \gamma(\rho) \coloneqq \operatorname{Tr}(\rho^2)=\sum_kp_k^2.
\end{equation}
\end{definition}

\begin{remark}[Ontic and epistemic uncertainty]
The distinction between pure and mixed states provides a useful conceptual entry point for this thesis. A pure state, $\rho=|\psi\rangle\langle\psi|$ and $\gamma=1$, displays \emph{ontic} uncertainty: even with complete knowledge of the preparation, incompatible quantum measurements have intrinsically probabilistic outcomes. A mixed state, $\gamma<1$, can additionally represent \emph{epistemic} uncertainty about a classical preparation. It can also arise by tracing out an inaccessible environment; this is an improper mixture and need not be interpreted as classical ignorance. Different ensemble decompositions can describe the same density operator, so the QFI depends on the state family $\rho(\theta)$ rather than on a unique ontic--epistemic decomposition. This distinction nevertheless helps separate intrinsic measurement randomness from uncertainty introduced by preparation and open-system dynamics.
\end{remark}

\subsection{Parameterized States and Measurements}

\begin{definition}[Parameterized quantum model]
The parameter of interest is encoded in an initial state $\rho_0$ through a parameter-dependent quantum channel,
\begin{equation}
    \rho(\theta)=\mathcal{E}_\theta(\rho_0)
    =\sum_n K_n(\theta)\rho_0K_n^\dagger(\theta),
\end{equation}
where $\sum_nK_n^\dagger K_n=\mathbb{I}$ for a trace-preserving channel. This description covers the open-system dynamics used in the noise analysis and the weak optical field model used in the astrophysical application.
\end{definition}

\begin{definition}[Positive operator-valued measure]
Information is extracted with a positive operator-valued measure (POVM) $\{M_x\}$, where
\begin{equation}
    M_x\geq0,\qquad \sum_xM_x=\mathbb{I},\qquad
    p(x|\theta)=\operatorname{Tr}[\rho(\theta)M_x].
    \label{eq:povm_def}
\end{equation}
Projective measurements and more general POVMs are discussed in Appendix~\ref{app:mathematical_background}.
\end{definition}

\section{From Classical to Quantum Fisher Information}

\begin{definition}[Classical Fisher information]
For a probability distribution $p(x|\theta)$, the classical Fisher information (CFI) is~\cite{Intro_Passage17}
\begin{equation}
    \mathcal{F}_{\mathrm{C}}(\theta)\coloneqq
    \int_{\mathcal{X}}\frac{[\partial_\theta p(x|\theta)]^2}{p(x|\theta)}\,\mathrm{d}x.
    \label{eq:classical_fi}
\end{equation}
\end{definition}

\begin{remark}[Operational meaning of CFI]
For $N$ independent samples, every locally unbiased estimator satisfies the classical Cramér--Rao bound,
\begin{equation}
    \operatorname{Var}(\hat{\theta})\geq\frac{1}{N\mathcal{F}_{\mathrm{C}}(\theta)}.
\end{equation}

The CFI depends on the chosen measurement. Quantum Fisher information is the maximum local information available over all allowed measurements, and therefore gives a measurement-independent benchmark for the state family $\rho(\theta)$.
\end{remark}

\section{Mixed-State Quantum Fisher Information}

\subsection{Symmetric Logarithmic Derivative}

\begin{definition}[SLD, QFI, and QFIM]
For a differentiable mixed-state trajectory $\rho(\theta)$, the symmetric logarithmic derivative (SLD) $L_\theta$ is defined on the support of $\rho$ by~\cite{Intro_Passage22}
\begin{equation}
    \partial_\theta\rho(\theta)
    =\frac{1}{2}\{\rho(\theta),L_\theta\}.
    \label{eq:sld_lyapunov}
\end{equation}
The scalar QFI is
\begin{equation}
    \mathcal{F}_Q(\theta)
    \coloneqq\operatorname{Tr}[\rho(\theta)L_\theta^2].
    \label{eq:qfi_sld_def}
\end{equation}
For a vector of physical parameters $\boldsymbol{\theta}$, the QFIM is
\begin{equation}
    H_{ij}(\boldsymbol{\theta})
    \coloneqq\frac{1}{2}\operatorname{Tr}\!\left[\rho
    (L_iL_j+L_jL_i)\right],
    \qquad L_i\equiv L_{\theta_i}.
    \label{eq:qfim_sld_def}
\end{equation}
Thus $\mathcal{F}_Q(\theta)=H_{\theta\theta}$ for a one-parameter model.
\end{definition}

\subsection{Spectral and Vectorized Evaluation}

\begin{remark}[Exact QFI evaluation]
If $\rho=\sum_k\lambda_k|\psi_k\rangle\langle\psi_k|$, the SLD matrix elements are
\begin{equation}
    \langle\psi_k|L_\theta|\psi_l\rangle
    =\frac{2\langle\psi_k|\partial_\theta\rho|\psi_l\rangle}
    {\lambda_k+\lambda_l},
    \qquad \lambda_k+\lambda_l>0,
    \label{eq:sld_spectral_elements}
\end{equation}
which yields
\begin{equation}
    \mathcal{F}_Q(\theta)
    =2\sum_{k,l:\lambda_k+\lambda_l>0}
    \frac{|\langle\psi_k|\partial_\theta\rho|\psi_l\rangle|^2}
    {\lambda_k+\lambda_l}.
    \label{eq:qfi_spectral_explicit}
\end{equation}
This formula is exact but requires diagonalizing a $d\times d$ density matrix, with cost $\mathcal{O}(d^3)$.

For the small systems used to obtain classical reference values, the QFIM can instead be evaluated through the support-restricted vectorized formula
\begin{equation}
    H_{ij}
    =2\,\operatorname{vec}[\partial_i\rho]^\dagger
    \left(\rho\otimes\mathbb{I}+\mathbb{I}\otimes\rho^T\right)^+
    \operatorname{vec}[\partial_j\rho],
    \label{eq:safranek_vectorized}
\end{equation}
where $+$ denotes the Moore--Penrose pseudoinverse on the relevant support. This is the formula used for the pre-detector QFI calculation in Chapter~\ref{sec:astrophysical_metrology_spade}.
\end{remark}

\section{Fidelity, Measurement Optimization, and Precision Bounds}

\begin{definition}[Root fidelity and Bures metric]
The root Uhlmann fidelity and Bures distance are~\cite{Intro_Passage44}
\begin{equation}
    F(\rho,\sigma)\coloneqq
    \operatorname{Tr}\!\left[\sqrt{\sqrt{\rho}\sigma\sqrt{\rho}}\right],
    \qquad
    d_B^2(\rho,\sigma)\coloneqq2-2F(\rho,\sigma).
    \label{def:bures_distance}
\end{equation}
For infinitesimally separated states, the Bures metric is related to the QFIM by
\begin{equation}
    H_{ij}=4g_{ij}^{B}.
    \label{def:qfim_bures_relation}
\end{equation}
\end{definition}

\begin{remark}[Why fidelity is used in this thesis]
This relation makes fidelity the natural starting point for the TQFI bounds introduced in Chapter~\ref{ch:methods}.
\end{remark}

\begin{theorem}[Braunstein--Caves optimization]
The Braunstein--Caves optimization identifies QFI as the maximal CFI over POVMs~\cite{Intro_Passage36},
\begin{equation}
    \mathcal{F}_Q(\theta)=\sup_{\{M_x\}}\mathcal{F}_{\mathrm{C}}(\theta;\{M_x\}),
\end{equation}
\end{theorem}

\begin{remark}[Quantum Cramér--Rao bound]
The theorem therefore implies the quantum Cramér--Rao bound
\begin{equation}
    \operatorname{Var}(\hat{\theta})
    \geq\frac{1}{N\mathcal{F}_Q(\theta)}.
    \label{eq:qcrb}
\end{equation}
The QCRB is the ideal precision benchmark; practical receivers and variational bounds need not attain it.
\end{remark}

\section{Ansatz Geometry and the Quantum Natural Gradient}

\begin{definition}[Ansatz QFIM]
The QFI used to assess a physical sensor must be distinguished from the QFIM used to optimize a variational circuit: the two may use the same mathematical construction, but refer to different parameters and different tasks. Let $\rho_{\boldsymbol{\alpha}}$ be the state prepared by an ansatz, with circuit parameters $\boldsymbol{\alpha}$. Its \emph{ansatz QFIM} is defined by the SLDs $\ell_\mu$ satisfying $\partial_\mu\rho_{\boldsymbol{\alpha}}=\{\rho_{\boldsymbol{\alpha}},\ell_\mu\}/2$:
\begin{equation}
    F^{\mathrm{ans}}_{\mu\nu}
    \coloneqq\frac{1}{2}\operatorname{Tr}\!\left[
    \rho_{\boldsymbol{\alpha}}(\ell_\mu\ell_\nu+\ell_\nu\ell_\mu)\right].
    \label{eq:ansatz_qfim}
\end{equation}
For a pure ansatz $\rho_{\boldsymbol{\alpha}}=|\psi(\boldsymbol{\alpha})\rangle\langle\psi(\boldsymbol{\alpha})|$, it reduces to four times the Fubini--Study metric,
\begin{equation}
    F^{\mathrm{ans}}_{\mu\nu}=4g^{\mathrm{FS}}_{\mu\nu},
    \qquad
    g^{\mathrm{FS}}_{\mu\nu}
    =\operatorname{Re}\!\left[
    \langle\partial_\mu\psi|\partial_\nu\psi\rangle
    -\langle\partial_\mu\psi|\psi\rangle
     \langle\psi|\partial_\nu\psi\rangle\right].
    \label{eq:fubini_study_def}
\end{equation}
Thus $F^{\mathrm{ans}}$ is a metric on circuit-parameter space, whereas $H_{ij}$ and $\mathcal{F}_Q(\theta)$ characterize information about physical parameters $\boldsymbol{\theta}$ and $\theta$, respectively.
\end{definition}

\begin{remark}[Quantum natural gradient]
For a variational loss $\mathcal{L}(\boldsymbol{\alpha})$, quantum natural-gradient descent uses the regularized pseudoinverse of the ansatz QFIM:
\begin{equation}
    \boldsymbol{\alpha}_{t+1}
    =\boldsymbol{\alpha}_t
    -\eta\left(F^{\mathrm{ans}}(\boldsymbol{\alpha}_t)
    +\lambda\mathbb{I}\right)^+
    \nabla_{\boldsymbol{\alpha}}\mathcal{L}(\boldsymbol{\alpha}_t),
    \qquad \lambda\geq0.
    \label{eq:qng_update}
\end{equation}
Here $\eta$ is the learning rate, $\lambda$ is a regularization parameter, and $+$ denotes the Moore--Penrose pseudoinverse. The update rescales ordinary gradients by the local distinguishability of ansatz states, rather than applying an inverse Hessian of the loss. Chapter~\ref{sec:vqse_deep_dive} discusses the resulting resource cost and stochastic approximations.
\end{remark}

\section{Connection to the Variational Algorithms}

Exact mixed-state QFI evaluation is exponentially costly for an $n$-qubit state because $d=2^n$. Chapter~\ref{ch:methods} therefore introduces dynamics-agnostic fidelity bounds (TQFI and SSQFI) and the VQSE/VQFIE subroutines used to estimate them on near-term hardware. The present chapter supplies their information-geometric motivation; their circuits, costs, and resource requirements are treated there rather than repeated here.
