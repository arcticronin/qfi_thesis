    \note{\scriptsize
        \textbf{Workflow formulas (reference).}\par
        $H(h_x)=-J\sum_j Z_jZ_{j+1}-h_x\sum_j X_j$,\quad $U_{h_x}(t)=e^{-itH(h_x)}$.\par
        $\rho_{\rm G}=\proj{\psi_0}$,\quad $\rho_\beta=e^{-\beta H}/Z_\beta$,\quad $\rho(t)=U\rho_{\rm in}U^\dagger$.\par
        $\rho_S=\Tr_E\rho$,\quad $\rho_S=\sum_i\lambda_i\proj{v_i}$,\quad $\rho_m=\sum_{i=1}^m\lambda_i\proj{v_i}$.\par
        $V\rho_S V^\dagger\approx\operatorname{diag}(\lambda_i)$; $V^\dagger\ket{i}$ recovers eigenvectors.\par
        Circuit gates approximate $U$; Thermal preparation is a separate process.
        The numerical and circuit methods share interfaces, not necessarily approximation errors.
        Finite-displacement fidelity bounds approach the local-QFI limit under regularity assumptions.}
