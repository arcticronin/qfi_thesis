\begin{frame}[plain,label=embedding]
    \begin{canvas}
        \heading{CHANNEL FRAMEWORK}{Classical probabilities and quantum states}{A probability vector has a diagonal quantum representation in a chosen basis.}
        \panel{.8}{6.18}{8.5}{3.55} \panel{8.75}{6.18}{15.2}{3.55}
        \node[txt,text=Teal,font=\small\bfseries] at (1.15,5.85) {CLASSICAL PROBABILITY EMBEDDING};
        \node[txt,font=\fontsize{15}{19}\selectfont] at (1.15,5.08)
        {$\boldsymbol p=\begin{pmatrix}p_0\\p_1\end{pmatrix}
                \;\longmapsto\;\rho_{\rm diag}=\begin{pmatrix}p_0&0\\0&p_1\end{pmatrix}$};
        \node[txt,text=Coral,font=\small\bfseries] at (9.1,5.85) {QUANTUM COHERENCE};
        \node[txt,font=\fontsize{17}{21}\selectfont] at (9.1,5.08)
        {$\rho=\begin{pmatrix}p_0&c\\c^*&p_1\end{pmatrix}$};
        \node[txt,font=\large] at (.85,3.09) {Classical channel: $\boldsymbol p'=R\boldsymbol p$};
        \node[txt,font=\large] at (8.95,3.09) {Quantum channel: $\rho'=\mathcal E(\rho)$};
        \node[smalltxt,text width=7cm] at (.85,2.52) {Redistributes probabilities between outcomes.};
        \node[smalltxt,text width=6.1cm] at (8.95,2.52) {Acts on populations and coherences.};
        \takeaway{The embedding preserves measured probabilities; coherences require further data.}
        \source{Two-outcome illustration: $p_i\geq0$, $p_0+p_1=1$, $|c|^2\leq p_0p_1$. A state vector of probabilities does not determine a quantum channel.}
        \footer{08 / 12}
    \end{canvas}
    \note{07:00--08:05. A classical probability vector is a state description. Put its entries on a diagonal to obtain a valid density operator with exactly those basis probabilities. A general quantum state also has off-diagonal coherences, which allow interference and are not inferred by this embedding. Then distinguish transformations: a stochastic matrix R acts on probabilities; a quantum channel E acts on the whole density operator. This lets us include coherent optics in the forward model without pretending the counts identify the missing coherences.}
\end{frame}
